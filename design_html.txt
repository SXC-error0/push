        function switchTab(el, type) {
            // Reset all tabs
            const buttons = el.parentElement.querySelectorAll('button');
            buttons.forEach(btn => {
                btn.classList.remove('text-primary', 'border-primary');
                btn.classList.add('text-on-surface-variant', 'border-transparent');
            });
            
            // Set active tab
            el.classList.add('text-primary', 'border-primary');
            el.classList.remove('text-on-surface-variant', 'border-transparent');
            
            // Interaction feedback
            el.style.transform = 'scale(0.95)';
            setTimeout(() => el.style.transform = 'scale(1)', 100);
        }

        // Add a bit of parallax floating to decorative icons
        document.addEventListener('mousemove', (e) => {
            const amount = 5;
            const x = (e.clientX / window.innerWidth - 0.5) * amount;
            const y = (e.clientY / window.innerHeight - 0.5) * amount;
            
            document.querySelectorAll('.sticker-rotate-1, .sticker-rotate-2').forEach(card => {
                card.style.transform = `rotate(${card.classList.contains('sticker-rotate-1') ? -1 : 1.5}deg) translate(${x}px, ${y}px)`;
            });
        });
    </script>

        function selectOption(element) {
            // Remove selection from others
            document.querySelectorAll('button').forEach(btn => {
                btn.classList.remove('bg-primary-container/20', 'border-primary', 'hard-shadow-active');
            });
            // Add selection to clicked
            element.classList.add('bg-primary-container/20', 'border-primary', 'hard-shadow-active');
            
            // Decorative effect
            const rect = element.getBoundingClientRect();
            createBurst(rect.left + rect.width / 2, rect.top + rect.height / 2);
        }

        function createBurst(x, y) {
            for (let i = 0; i < 8; i++) {
                const particle = document.createElement('div');
                particle.innerHTML = '✨';
                particle.className = 'fixed pointer-events-none text-xl z-50';
                particle.style.left = x + 'px';
                particle.style.top = y + 'px';
                document.body.appendChild(particle);

                const angle = (i / 8) * Math.PI * 2;
                const velocity = 4 + Math.random() * 4;
                const vx = Math.cos(angle) * velocity;
                const vy = Math.sin(angle) * velocity;

                let opacity = 1;
                let currentX = x;
                let currentY = y;

                function animate() {
                    currentX += vx;
                    currentY += vy;
                    opacity -= 0.02;
                    particle.style.left = currentX + 'px';
                    particle.style.top = currentY + 'px';
                    particle.style.opacity = opacity;

                    if (opacity > 0) {
                        requestAnimationFrame(animate);
                    } else {
                        particle.remove();
                    }
                }
                animate();
            }
        }
    </script>
